## new 操作符底层做了什么

```javascript
function myNew(fn, ...args) {
  // 创建一个空对象
  const obj = {}
  // 对象的 __proto__ 指向构造函数的原型对象
  obj.__proto__ = fn.prototype
  // 原函数通过 apply 把 this 绑定到新对象 obj
  const result = fn.apply(obj, args)

  return result instanceof Object ? result : obj
}
```

## 原型链的查找过程

```javascript
1、先找对象自身
obj.hasOwnProperty('xxx')
2、再找原型
obj.__proto__
3、继续往上查找
obj.__proto__.__proto__
4、直到 null
Object.prototype.__proto__ === null
```

## ES6 class 之前最优继承方式

```javascript
function inherit(child, parent) {

  child.prototype = Object.create(parent.prototype)

  child.prototype.constructor = child
}

inherit(Dog, Animal)
```

## 经典题

```javascript
function Animal() {
  this.colors = ['black', 'white']
}

function Dog() {}

// 共享同一个原型对象
Dog.prototype = new Animal()

const d1 = new Dog()
const d2 = new Dog()

d1.colors.push('red')

console.log(d2.colors)// ['black','white','red'] 共享一个原型对象
```


```javascript
function Foo() {}

const f = new Foo()

console.log(f.__proto__ === Foo.prototype) // true
```

```javascript
function Foo(){}

Foo.prototype.a = 1

const f = new Foo()

Foo.prototype.a = 2

console.log(f.a) // 2
```

```javascript
function Foo(){}

Foo.prototype.a = 1

const f1 = new Foo()

Foo.prototype = { a:2 }

const f2 = new Foo()

console.log(f1.a) // 1
console.log(f2.a) // 2
```

```javascript
function Foo(){}

Foo.prototype.a = 1

const f = new Foo()

f.a = 10

console.log(f.a) // 10
delete f.a

console.log(f.a) // 1 删除自身属性后，会从原型链查找
```

```javascript
function Foo(){}

Foo.prototype = {
  constructor: Foo,
  a:1
}

const f = new Foo()

console.log(f.constructor === Foo) // true
```
