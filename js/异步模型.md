# 异步模型


## 经典题

```javascript
console.log(1)

// 宏任务
setTimeout(() => {
  console.log(2)
}, 0)

console.log(3)

// 1 3 2
```

`then` 返回一个新的 Promise.
```javascript
Promise.resolve(1)
  .then(x => x + 1)
  .then(x => console.log(x)) // 2
```

`async`等价于 返回 Promise。

```javascript
async function foo() {
  return 10
}

foo().then(console.log) // 10
```


同步任务：1 2 4， 微任务 3
```javascript
console.log(1)

new Promise(resolve => {
  console.log(2)
  resolve()
})
.then(() => {
  console.log(3)
})

console.log(4)

// 1 2 4 3
```

```javascript
// 同步任务
console.log(1)

// 宏任务
setTimeout(() => {
  console.log(2)
}, 0)

Promise.resolve().then(() => {
  // 微任务
  console.log(3)
})

// 同步任务
console.log(4)
// 1 4 3 2
```



```javascript
async function foo() {
  // 同步任务
  console.log(1)

  await Promise.resolve()
  // await 之后进入 microtask
  console.log(2)
}

// 同步任务
console.log(3)

foo()

// 同步任务
console.log(4)

// 3 1 4 2
```


```javascript
// 同步任务
console.log(1)

// 宏任务
setTimeout(() => {
  console.log(2)
}, 0)

Promise.resolve()
  .then(() => {
    // 微任务
    console.log(3)
  })
  .then(() => {
    // 微任务
    console.log(4)
  })

// 同步任务
console.log(5)
// 1 5 3 4 2
```


```javascript
Promise.resolve()
  .then(() => {
  // 相当于。return Promise.reject()
    throw new Error("error")
  })
  .then(() => {
    console.log("success")
  })
  .catch(err => {
    console.log("catch")
  })
// catch
```

```javascript
async function foo() {
  throw new Error("fail")
}

foo().catch(() => {
  console.log("error")
})
// error
```
