# 类与继承

| 维度      | ES5              | ES6     |
| ------- | ---------------- | ------- |
| 继承方式    | prototype + call | extends |
| this 创建 | 子类自己创建           | 父类创建    |
| super   | 没有               | 有       |
| 语法复杂度   | 高                | 低       |
| 本质      | 原型链              | 原型链     |


## ES5继承的写法

```javascript
function Animal(name) {
  this.name = name
}
// 方法写在原型对象上
Animal.prototype.say = function() {}

function Dog(name) {
  // call + this 指定
  Animal.call(this, name)
}

// 创建新对象，绑定原型
Dog.prototype = Object.create(Animal.prototype)

Dog.prototype.constructor = Dog
```

## ES6继承

```javascript
class Animal {
  constructor(name) {
    this.name = name
  }

  say() {}
}

class Dog extends Animal {
  constructor(name) {
    super(name)
  }
}
```

babel 编译后的核心代码
```javascript
function _inherits(subClass, superClass) {
  subClass.prototype = Object.create(superClass.prototype)
  subClass.prototype.constructor = subClass
  // 静态属性继承
  Object.setPrototypeOf(subClass, superClass)
}
```
## 经典题


```javascript
class A {}
class B extends A {}

console.log(B.__proto__ === A) // true
console.log(B.prototype.__proto__ === A.prototype) // true
```

```javascript
class A {
  constructor() {
    this.x = 1
  }
}

class B extends A {
  constructor() {
    console.log(this) // ReferenceError super() 之前不能用 this
    super()
  }
}

new B()
```

```javascript
class A {
  constructor() {
    this.a = 1
  }

  foo() {
    console.log("A")
  }
}

class B extends A {
  foo() {
    super.foo() // "A"
    console.log("B") "B"
  }
}

new B().foo()
```

```javascript
class A {}
class B extends A {}

const b = new B()

console.log(b.__proto__ === B.prototype) // true
console.log(B.prototype.__proto__ === A.prototype) // true
console.log(B.__proto__ === A) // true
```
