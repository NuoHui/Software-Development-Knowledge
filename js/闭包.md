# 闭包

闭包本质：函数 + 创建时的词法环境。

```javascript
// 所有闭包共享同一个 i，i 在 global scope. 可以改为 let
for (var i = 0; i < 3; i++) {
  setTimeout(function() {
    console.log(i) // 3 3 3
  }, 0)
}

// 解决方法
for (var i = 0; i < 3; i++) {
  (function(i) {
    setTimeout(function() {
      console.log(i)
    })
  })(i)
}
```

```javascript
function foo() {
  var a = 1

  return function() {
    console.log(a)
  }
}

const fn = foo()
fn() // 1
```

```javascript

function foo() {
  let a = 1

  return function() {
    a++
    console.log(a)
  }
}

const fn1 = foo()
const fn2 = foo()

fn1() // 2
fn1() // 3
fn2() // 2
// 每次 foo() 创建新的闭包
```

```javascript
function foo() {
  let a = 1

  return {
    inc() {
      a++
    },
    log() {
      console.log(a)
    }
  }
}

const obj = foo()

obj.inc()
obj.inc()
obj.log() // 3
```
